
# Personalism/Pudgala Thesis
tenet_rice_personalism = {
	icon = tenet_rice_personalism

	is_shown = {
		religion_tag = buddhism_religion
	}

	piety_cost = {
		if = {
			limit = {
				always = no
			}
			value = faith_tenet_cost_low
		}
		else = {
			value = faith_tenet_cost_massive
		}

		# Multiplier for keeping same tenet
		if = {
			limit = { rite_has_tenet = tenet_rice_personalism }
			multiply = faith_unchanged_doctrine_cost_mult
		}
	}

	special_parameters = {
		hostility_levels = {
			same_religion = astray
		}
	}

	character_modifier = {
		learning_per_piety_level = 1
		stress_loss_per_piety_level = 0.05
	}

	personal_tenet_modifier = {
		character_spiritual_fulfillment_gain_mult = 0.1
		character_spiritual_fulfillment_loss_mult = -0.1
	}
}
